package com.example;

import com.mongodb.client.MongoClient;
import com.mongodb.client.MongoClients;
import com.mongodb.client.MongoDatabase;
import com.mongodb.client.MongoCollection;
import com.mongodb.client.model.Filters;
import com.mongodb.client.model.Updates;
import org.bson.Document;
import org.bson.types.ObjectId;
import io.github.cdimascio.dotenv.Dotenv;
import java.util.Scanner;

public class App {

    public static void main(String[] args) {

        Dotenv dotenv = Dotenv.configure()
                .directory("D:/JAVA MONGODB CONN/mongodb-java-demo")
                .ignoreIfMissing()
                .load();

        String uri = dotenv.get("MONGODB_URI");

        if (uri == null || uri.isBlank()) {
            System.out.println("Please set MONGODB_URI in your .env file.");
            return;
        }

        try (
            MongoClient client = MongoClients.create(uri);
            Scanner sc = new Scanner(System.in)
        ) {
            MongoDatabase database = client.getDatabase("collegeDB");
            MongoCollection<Document> students =
                    database.getCollection("students");

            database.runCommand(new Document("ping", 1));
            System.out.println("MongoDB Atlas connected successfully!");

            while (true) {
                System.out.println("\n--- Student Management Menu ---");
                System.out.println("1. Add Student");
                System.out.println("2. View Students");
                System.out.println("3. Update Student");
                System.out.println("4. Delete Student");
                System.out.println("5. Exit");
                System.out.print("Enter choice: ");

                if (!sc.hasNextInt()) {
                    System.out.println("Invalid input! Enter a number.");
                    sc.nextLine();
                    continue;
                }

                int choice = sc.nextInt();
                sc.nextLine();

                switch (choice) {

                    case 1: {
                        System.out.print("Enter student name: ");
                        String name = sc.nextLine();

                        System.out.print("Enter age: ");
                        if (!sc.hasNextInt()) {
                            System.out.println("Age must be a number!");
                            sc.nextLine();
                            break;
                        }
                        int age = sc.nextInt();
                        sc.nextLine();

                        System.out.print("Enter course: ");
                        String course = sc.nextLine();

                        Document student = new Document("name", name)
                                .append("age", age)
                                .append("course", course);

                        students.insertOne(student);
                        System.out.println("Student added successfully!");
                        System.out.println("Student ID: "
                                + student.getObjectId("_id"));
                        break;
                    }

                    case 2: {
                        System.out.println("\n--- Student Records ---");

                        boolean found = false;

                        for (Document student : students.find()) {
                            System.out.println(student.toJson());
                            found = true;
                        }

                        if (!found) {
                            System.out.println("No students found.");
                        }
                        break;
                    }

                    case 3: {
                        System.out.print("Enter student ID to update: ");
                        String id = sc.nextLine().trim();

                        if (!ObjectId.isValid(id)) {
                            System.out.println("Invalid student ID!");
                            break;
                        }

                        ObjectId objectId = new ObjectId(id);

                        Document existing = students.find(
                                Filters.eq("_id", objectId)
                        ).first();

                        if (existing == null) {
                            System.out.println("Student not found!");
                            break;
                        }

                        System.out.println("Current record: "
                                + existing.toJson());

                        System.out.print("Enter new name: ");
                        String name = sc.nextLine();

                        System.out.print("Enter new age: ");
                        if (!sc.hasNextInt()) {
                            System.out.println("Age must be a number!");
                            sc.nextLine();
                            break;
                        }
                        int age = sc.nextInt();
                        sc.nextLine();

                        System.out.print("Enter new course: ");
                        String course = sc.nextLine();

                        students.updateOne(
                                Filters.eq("_id", objectId),
                                Updates.combine(
                                        Updates.set("name", name),
                                        Updates.set("age", age),
                                        Updates.set("course", course)
                                )
                        );

                        System.out.println("Student updated successfully!");
                        break;
                    }

                    case 4: {
                        System.out.print("Enter student ID to delete: ");
                        String id = sc.nextLine().trim();

                        if (!ObjectId.isValid(id)) {
                            System.out.println("Invalid student ID!");
                            break;
                        }

                        ObjectId objectId = new ObjectId(id);

                        Document existing = students.find(
                                Filters.eq("_id", objectId)
                        ).first();

                        if (existing == null) {
                            System.out.println("Student not found!");
                            break;
                        }

                        System.out.println("Student to delete: "
                                + existing.toJson());
                        System.out.print("Are you sure? (yes/no): ");

                        String confirm = sc.nextLine().trim();

                        if (confirm.equalsIgnoreCase("yes")) {
                            students.deleteOne(
                                    Filters.eq("_id", objectId)
                            );
                            System.out.println("Student deleted successfully!");
                        } else {
                            System.out.println("Deletion cancelled.");
                        }
                        break;
                    }

                    case 5:
                        System.out.println("Exiting program...");
                        return;

                    default:
                        System.out.println("Invalid choice!");
                }
            }

        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
        }
    }
}

package com.example;

import com.mongodb.client.MongoClient;
import com.mongodb.client.MongoClients;
import com.mongodb.client.MongoCollection;
import com.mongodb.client.MongoDatabase;
import com.mongodb.client.model.Filters;
import org.bson.Document;
import io.github.cdimascio.dotenv.Dotenv;

import java.util.Scanner;

public class App {
    public static void main(String[] args) {

        Dotenv dotenv = Dotenv.configure()
                .directory("D:/JAVA MONGODB CONN/mongodb-java-demo")
                .ignoreIfMissing()
                .load();

        String uri = dotenv.get("MONGODB_URI");

        if (uri == null || uri.isBlank()) {
            System.out.println("MONGODB_URI not found in .env file.");
            return;
        }

        try (
            MongoClient client = MongoClients.create(uri);
            Scanner sc = new Scanner(System.in)
        ) {
            MongoDatabase database = client.getDatabase("collegeDB");
            MongoCollection<Document> students =
                    database.getCollection("students");

            database.runCommand(new Document("ping", 1));
            System.out.println("MongoDB connected successfully!");

            while (true) {
                System.out.println("\n--- Student Search Menu ---");
                System.out.println("1. Search by Student ID");
                System.out.println("2. Search by Department");
                System.out.println("3. Search by Minimum Marks");
                System.out.println("4. Search by Skill");
                System.out.println("5. View All Students");
                System.out.println("6. Exit");
                System.out.print("Enter choice: ");

                if (!sc.hasNextInt()) {
                    System.out.println("Invalid choice!");
                    sc.nextLine();
                    continue;
                }

                int choice = sc.nextInt();
                sc.nextLine();

                switch (choice) {
                    case 1: {
                        System.out.print("Enter student ID: ");

                        if (!sc.hasNextInt()) {
                            System.out.println("Invalid student ID!");
                            sc.nextLine();
                            break;
                        }

                        int studentId = sc.nextInt();
                        sc.nextLine();

                        Document student = students.find(
                                Filters.eq("student_id", studentId)
                        ).first();

                        if (student != null) {
                            System.out.println(student.toJson());
                        } else {
                            System.out.println("Student not found.");
                        }
                        break;
                    }

                    case 2: {
                        System.out.print("Enter department: ");
                        String department = sc.nextLine().trim();

                        showStudents(
                                students,
                                Filters.eq("department", department)
                        );
                        break;
                    }

                    case 3: {
                        System.out.print("Enter minimum marks: ");

                        if (!sc.hasNextInt()) {
                            System.out.println("Invalid marks!");
                            sc.nextLine();
                            break;
                        }

                        int marks = sc.nextInt();
                        sc.nextLine();

                        showStudents(
                                students,
                                Filters.gte("marks", marks)
                        );
                        break;
                    }

                    case 4: {
                        System.out.print("Enter skill: ");
                        String skill = sc.nextLine().trim();

                        showStudents(
                                students,
                                Filters.eq("skills", skill)
                        );
                        break;
                    }

                    case 5:
                        showStudents(students, new Document());
                        break;

                    case 6:
                        System.out.println("Exiting program...");
                        return;

                    default:
                        System.out.println("Invalid choice!");
                }
            }

        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
        }
    }

    private static void showStudents(
            MongoCollection<Document> students,
            org.bson.conversions.Bson filter) {

        boolean found = false;

        for (Document student : students.find(filter)) {
            System.out.println(student.toJson());
            found = true;
        }

        if (!found) {
            System.out.println("No matching students found.");
        }
    }
}
